### 0.3 资源边界

规格规定数值进入数学前夹取，未给具体容量。实现统一在后端边界定义常量，前端使用正常可编辑范围，不复制安全算法：

| 输入 | 规则 |
|---|---|
| x/y | 数学输入夹至 -1,000,000…1,000,000；负数仍被保存闸明确拒绝，不夹成0掩盖错误 |
| w/h | 数学输入夹至0…1,000,000 |
| 声明 mulX/mulY | 0表示兼容回落，非0夹至1…8；负数按1 |
| 声明 cell | 0表示旧 qrCell(size)，非0夹至2…8 |
| size | 数学输入上界4096；非正值仍走既有各类型默认，不能把旧 size 推导倍率无条件夹到8 |
| elements JSON | UTF-8不超过262,144字节、元素不超过128个；超限不进入布局，保存400、resolve返回index=-1容量issue和空labels；文案分别「模板元素 JSON 超过 256 KiB」「模板元素不能超过 128 个」 |
| 单元素解析后内容 | 不超过4096个 codepoint；超限400「第 N 个元素内容超过 4096 字符」，不截断业务字段冒充完整内容 |
| 纸头 | 先沿用正宽高/正dpi校验及原顺序，再用long检查换算后宽高各不超过1,000,000点；超限400「标签画布不能超过 1000000 点」 |

容量限制属于对外请求防耗尽；正常模板与已冻结输入保持原样。上线前只读盘点已有模板是否超出新增资源边界；有命中就暂停发布并报告，不批量改库。类型错误、非数组 JSON、null 元素必须返回可读400/issues，不能500；浮点坐标不默默截断，拒绝为「坐标必须是整数点」，UI提交前主动取整。

```java
    return content == null ? "" : content.replace("\"", "'")
        .replace("\r", " ").replace("\n", " ").trim();
}

public static List<Box> segmentsFor(ResolvedElement e, int paperW) {
    String content = sanitize(e.getContent());
    if (content.isEmpty()) return List.of();
    LabelGeometry.Font font = LabelGeometry.fontFor(content, e.getFontSize());
    int available = LabelGeometry.wrapBudgetDots(e.getX(), e.getW(), paperW);
    List<String> lines = LabelGeometry.lineSegments(content, font.perCharDots(), available);
    List<Box> result = new ArrayList<>(lines.size());
    for (int i = 0; i < lines.size(); i++) {
        String line = lines.get(i);
        result.add(new Box(e.getX(), e.getY() + i * font.lineHeightDots(),
            line.codePointCount(0, line.length()) * font.perCharDots(),
            font.lineHeightDots(), line));
    }
    return result;
}
```

此步只给text/multiline调用；Task 3加类型分支和统一输入边界。`textBlock`从盒子取x/y/content，字体仍同一font函数；粗体继续发第二条x+1，不把它计成多一行。非文本分支仍使用同一sanitize，BARCODE/QRCODE形参不动。

```java
@Transactional(rollbackFor = Exception.class)
public void update(BaseLabelTemplate next) {
    permService.requirePerm(PERM);
    if (next.getId() == null) throw BizException.of(400, "缺少模板 id");
    BaseLabelTemplate old = mapper.selectForUpdate(next.getId());
    if (old == null) throw BizException.of(404,
        "标签模板不存在或已删除: id=" + next.getId());
    next.setBizType(old.getBizType());
    next.setIsDefault(old.getIsDefault());
    applyDefaults(next);
    requireUniqueCode(next.getTemplateCode(), next.getId());
    gate(next);
    snapshotAndReplace(old, next);
}
```

```javascript
async function refreshGeometry(gen) {
  if (disposed || gen !== resolveGen) return
  const payload = buildResolveRequest()
  if (!payload) {
    resolving.value = false
    return
  }
  resolving.value = true

    const result = await labelApi.resolve(payload)
    if (disposed || gen !== resolveGen) return
    resolved.value = result.data
    resolvedGen.value = gen
```

```javascript
function resizePatch(row, box, deltaX, deltaY) {
  if (row.type === 'text' || row.type === 'multiline') {
    const segment = box.segments[0]
    if (!segment || !segment.w || !segment.h) return null
    const stepX = segment.w / box.mulX
    const stepY = segment.h / box.mulY
    const patch = {}
    if (deltaX) patch.mulX = Math.max(1, Math.min(8,
      box.mulX + Math.round(deltaX / stepX)))
    if (deltaY) patch.mulY = Math.max(1, Math.min(8,
      box.mulY + Math.round(deltaY / stepY)))
    return patch
  }
  if (row.type === 'qrcode') {

    return { cell: Math.max(2, Math.min(8,

}
```

```javascript
let history = []
let historyCursor = -1
let historyTimer
let restoringHistory = false
const rowsSnapshot = () => JSON.stringify(rows)

function recordRows() {
  clearTimeout(historyTimer)
  const snapshot = rowsSnapshot()
  if (history[historyCursor] === snapshot) return
  history = history.slice(0, historyCursor + 1)
  history.push(snapshot)
  if (history.length > 51) history.shift()
  historyCursor = history.length - 1
}

function resetHistory() {
  clearTimeout(historyTimer)
  history = [rowsSnapshot()]
  historyCursor = 0
}

function undoRows() {
  recordRows()
  if (historyCursor <= 0) return
  restoringHistory = true
  historyCursor--
  rows.splice(0, rows.length, ...JSON.parse(history[historyCursor]))
  restoringHistory = false
}
```
